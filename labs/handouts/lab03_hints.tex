\vspace{2pt}\hrule\vspace{4pt}

\textbf{Step 3 --- attach a fresh classification head.}\; {\small Load DistilBERT \emph{and} bolt a
brand-new 5-class head on top (one head per disease class). This is the whole idea of transfer
learning: keep the pretrained body, swap the head (Lecture 8).}

\textbf{Blank 1 --- build \texttt{model}.}
\begin{itemize}
  \item[(a)] \opt{model = AutoModelForSequenceClassification.from\_pretrained(CHECKPOINT, num\_labels=NUM\_LABELS)}
  \item[(b)] \opt{model = AutoModel.from\_pretrained(CHECKPOINT)}
  \item[(c)] \opt{model = AutoModelForSequenceClassification.from\_pretrained(CHECKPOINT)}
\end{itemize}
{\small (a) loads the body \emph{and} a new 5-way head. (b) is the body only --- no classifier, so it
cannot output class scores. (c) forgets \opt{num\_labels}, so it defaults to a \textbf{2}-class head and
the shape check fails (it wants \opt{[batch, 5]}).}

\vspace{3pt}\hrule\vspace{4pt}

\textbf{Step 3 --- the ``frozen body'' run.}\; {\small For the head-only comparison, freeze every weight
in the DistilBERT \emph{body} so only the new head learns. The body is \opt{model.distilbert}; the head
is \opt{model.classifier}.}

\textbf{Blank 2 --- freeze the body.}
\begin{itemize}
  \item[(a)] \opt{for param in model.distilbert.parameters():}\; \opt{param.requires\_grad = False}
  \item[(b)] \opt{for param in model.parameters():}\; \opt{param.requires\_grad = False}
  \item[(c)] \opt{for param in model.classifier.parameters():}\; \opt{param.requires\_grad = False}
\end{itemize}
{\small (a) freezes only the body $\Rightarrow$ the check sees exactly \textbf{594{,}437} trainable head
params. (b) freezes \emph{everything} --- nothing can learn. (c) freezes the head and leaves the body
trainable --- the opposite of what ``head-only'' means.}

\vspace{3pt}\hrule\vspace{4pt}

\textbf{Step 4 --- the fine-tuning learning rate.}\; {\small When you start from pretrained weights you
nudge them \emph{gently} --- a large step would erase what the model already knows. Fine-tuning uses a
much smaller rate than the from-scratch run (\opt{SCRATCH\_LR = 5e-4}). Lecture 4: the learning rate is
the single most important knob.}

\textbf{Blank 3 --- \texttt{FINETUNE\_LR}.}
\begin{itemize}
  \item[(a)] \opt{FINETUNE\_LR = 2e-5}
  \item[(b)] \opt{FINETUNE\_LR = 5e-4}
  \item[(c)] \opt{FINETUNE\_LR = 0.1}
\end{itemize}
{\small (a) is the usual transformer fine-tuning rate --- small, so the pretrained knowledge survives.
(b) equals the \emph{from-scratch} rate; the check wants fine-tuning to step more gently than that.
(c) is enormous for a pretrained model and blows the weights apart.}
